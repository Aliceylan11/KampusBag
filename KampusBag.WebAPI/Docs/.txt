-- ── ADIM 1: Test için kullanıcı UUID'sini buraya yapıştırın ──────────────
-- pgAdmin'de F5 ile çalıştırmadan önce bu satırı güncelleyin:

DO $setup$
BEGIN
    -- Geçici session değişkeni olarak userId ayarla
    PERFORM set_config(
        'kampusbag.current_user_id',
        'df3471bc-cf8c-4e36-8832-57bee62c00dc',  -- ← kendi userId'nizi yazın
        false
    );
END;
$setup$;

-- ── ADIM 2: Ana sorgu — current_setting() ile parametreyi okur ───────────
WITH

-- ──────────────────────────────────────────────────────────────────────────
-- CTE 1 · PRIVATE_THREADS
-- Kullanıcının birebir mesajlaştığı kişileri ve son mesajı çıkarır.
-- ──────────────────────────────────────────────────────────────────────────
private_threads AS (
    SELECT
        CASE
            WHEN m."SenderId" = current_setting('kampusbag.current_user_id')::uuid
                THEN m."ReceiverId"
            ELSE m."SenderId"
        END                                           AS contact_id,
        m."Content"                                   AS last_content,
        m."SentAt"                                    AS last_sent_at,
        m."IsSilent"                                  AS is_silent,
        m."IsEmergency"                               AS is_emergency,
        ROW_NUMBER() OVER (
            PARTITION BY
                LEAST(m."SenderId"::text, m."ReceiverId"::text),
                GREATEST(m."SenderId"::text, m."ReceiverId"::text)
            ORDER BY m."SentAt" DESC
        )                                             AS rn
    FROM "Messages" m
    WHERE
        m."CourseId" IS NULL
        AND (
            m."SenderId"   = current_setting('kampusbag.current_user_id')::uuid
            OR m."ReceiverId" = current_setting('kampusbag.current_user_id')::uuid
        )
),

-- ──────────────────────────────────────────────────────────────────────────
-- CTE 2 · PRIVATE_UNREAD
-- Her konuşma için okunmamış mesaj sayısı.
-- ──────────────────────────────────────────────────────────────────────────
private_unread AS (
    SELECT
        m."SenderId"  AS contact_id,
        COUNT(*)      AS unread_count
    FROM "Messages" m
    WHERE
        m."ReceiverId" = current_setting('kampusbag.current_user_id')::uuid
        AND m."IsRead"   = false
        AND m."CourseId" IS NULL
    GROUP BY m."SenderId"
),

-- ──────────────────────────────────────────────────────────────────────────
-- CTE 3 · PRIVATE_SUMMARY
-- Son mesaj + kişi bilgisi + okunmamış + Sessiz Mod bayrağı
-- ──────────────────────────────────────────────────────────────────────────
private_summary AS (
    SELECT
        'private'::text                                            AS chat_type,
        current_setting('kampusbag.current_user_id') || '_'
            || pt.contact_id::text                                 AS chat_id,
        u."FullName"                                               AS display_name,
        u."Id"                                                     AS other_user_id,
        u."Role"                                                   AS other_user_role,
        NULL::uuid                                                 AS course_id,
        pt.last_content                                            AS last_message,
        pt.last_sent_at,
        COALESCE(pu.unread_count, 0)                               AS unread_count,

        -- Sessiz Mod: Alıcı Akademisyen (Role=2) + saat 17:00 sonrası
        CASE
            WHEN u."Role" = 2
             AND EXTRACT(HOUR FROM NOW() AT TIME ZONE 'Europe/Istanbul') >= 17
            THEN true
            ELSE false
        END                                                        AS is_silent_mode,

        false                                                      AS is_locked,
        pt.is_emergency

    FROM private_threads pt
    INNER JOIN "Users" u
        ON u."Id" = pt.contact_id
    LEFT JOIN private_unread pu
        ON pu.contact_id = pt.contact_id
    WHERE pt.rn = 1
),

-- ──────────────────────────────────────────────────────────────────────────
-- CTE 4 · COURSE_LAST_MSG
-- Her ders kanalının en son mesajını seçer.
-- ──────────────────────────────────────────────────────────────────────────
course_last_msg AS (
    SELECT
        m."CourseId",
        m."Content"                                    AS last_content,
        m."SentAt"                                     AS last_sent_at,
        m."IsEmergency"                                AS is_emergency,
        ROW_NUMBER() OVER (
            PARTITION BY m."CourseId"
            ORDER BY m."SentAt" DESC
        )                                              AS rn
    FROM "Messages" m
    WHERE m."CourseId" IS NOT NULL
),

-- ──────────────────────────────────────────────────────────────────────────
-- CTE 5 · COURSE_UNREAD
-- Her ders kanalı için okunmamış mesaj sayısı.
-- ──────────────────────────────────────────────────────────────────────────
course_unread AS (
    SELECT
        m."CourseId",
        COUNT(*) AS unread_count
    FROM "Messages" m
    INNER JOIN "CourseMemberships" cm   -- IN yerine JOIN → daha hızlı
        ON cm."CourseId" = m."CourseId"
        AND cm."UserId"  = current_setting('kampusbag.current_user_id')::uuid
    WHERE
        m."SenderId" <> current_setting('kampusbag.current_user_id')::uuid
        AND m."IsRead"   = false
        AND m."CourseId" IS NOT NULL
    GROUP BY m."CourseId"
),

-- ──────────────────────────────────────────────────────────────────────────
-- CTE 6 · COURSE_SUMMARY
-- Resmi Kanal / Çalışma Odası sınıflandırması + son mesaj birleştirme
-- ──────────────────────────────────────────────────────────────────────────
course_summary AS (
    SELECT
        CASE
            WHEN cm."IsRepresentative" = false THEN 'official'::text
            ELSE 'study'::text
        END                                            AS chat_type,
        c."Id"::text                                   AS chat_id,
        c."Name"                                       AS display_name,
        NULL::uuid                                     AS other_user_id,
        NULL::int                                      AS other_user_role,
        c."Id"                                         AS course_id,
        COALESCE(clm.last_content, 'Henüz mesaj yok') AS last_message,
        COALESCE(
            clm.last_sent_at,
            '1970-01-01 00:00:00+00'::timestamptz
        )                                              AS last_sent_at,
        COALESCE(cu.unread_count, 0)                   AS unread_count,
        false                                          AS is_silent_mode,
        CASE
            WHEN cm."IsRepresentative" = false THEN true
            ELSE false
        END                                            AS is_locked,
        false                                          AS is_emergency
    FROM "CourseMemberships" cm
    INNER JOIN "Courses" c
        ON c."Id" = cm."CourseId"
    LEFT JOIN course_last_msg clm
        ON clm."CourseId" = c."Id" AND clm.rn = 1
    LEFT JOIN course_unread cu
        ON cu."CourseId" = c."Id"
    WHERE cm."UserId" = current_setting('kampusbag.current_user_id')::uuid
)

-- ══════════════════════════════════════════════════════════════════════════
-- SONUÇ: Private + Official + Study — en yeni üstte
-- ══════════════════════════════════════════════════════════════════════════
SELECT
    chat_type,
    chat_id,
    display_name,
    other_user_id,
    other_user_role,
    course_id,
    last_message,
    last_sent_at,
    unread_count,
    is_silent_mode,
    is_locked,
    is_emergency
FROM private_summary

UNION ALL

SELECT
    chat_type,
    chat_id,
    display_name,
    other_user_id,
    other_user_role,
    course_id,
    last_message,
    last_sent_at,
    unread_count,
    is_silent_mode,
    is_locked,
    is_emergency
FROM course_summary

ORDER BY last_sent_at DESC;
